func timerFunc() {
 ticker:= time.NewTicker(time.Duration(timerSeconds) * time.Second)
 defer ticker.Stop()

 currentWindow = windowName()
 previousTime = time.Now().Unix()
 fileFunc(0, strconv.FormatInt(previousTime, 10)+";"+strconv.FormatInt(previousTime, 10)+";"+currentWindow)

 for {
  select {
   case <-ticker.C:
    if debugOn {
     fmt.Println("tick")
    }
    currentTime := time.Now().Unix()
    tempWindow := windowName()
    if tempWindow == currentWindow {
     fileFunc(1, strconv.FormatInt(previousTime, 10)+";"+strconv.FormatInt(currentTime, 10)+";"+tempWindow)
    } else {
     fileFunc(0, strconv.FormatInt(currentTime, 10)+";"+strconv.FormatInt(currentTime, 10)+";"+tempWindow)
     previousTime = currentTime
     currentWindow = tempWindow
    }
   case <-stopTicker:
    ticker.Stop()
    return
  }
 }
}

package main

import (
 "fmt"
 "math"
)

var red = "\033[31m"
var green = "\033[32m"
var blue = "\033[34m"
var yellow = "\033[33m"
var magenta = "\033[35m"
var cyan = "\033[36m"
var reset = "\033[0m"

func main() {
  percentages := []float64{25, 15, 30, 10, 12, 111}
  colors := []string{red, yellow, green, cyan, blue, magenta}

  // Calculate total and boundaries
  total := 0.0
  for _, p := range percentages {
    total += p
  }

boundaries := make([]float64, len(percentages)+1)
boundaries[0] = 0 
  for i, p := range percentages {
    boundaries[i+1] = boundaries[i] + (p/total)*2*math.Pi
  }

  radius:=4
  expected:= radius * radius;
  for y:=-radius; y<=radius; y++{
   for x:=-radius*2; x<=radius*2; x++{
    value:= (x/2)*(x/2) + y*y
    if value <= expected {
     angle := math.Atan2(float64(y), float64(x/2))
     if angle < 0 {angle += 2 * math.Pi}

        colorIndex := 0
        for i := 0; i < len(boundaries)-1; i++ {
          if angle >= boundaries[i] && angle < boundaries[i+1] {
            colorIndex = i
            break
          }
        }
        
        fmt.Print(colors[colorIndex] + "█" + reset)
      } else {
        fmt.Print(" ")
      }
    }
    fmt.Printf("\n")
  }
}

func saveSettings(){
 settings := map[string]string{
  "private": fmt.Sprintf("%v", privateOn),
  "timezone": fmt.Sprintf("%d", timezone),
  "deleteOld": fmt.Sprintf("%d", deleteOld),
  "timer": fmt.Sprintf("%d", timerSeconds),
  "logAmount": fmt.Sprintf("%d", logAmount),
  "includes": strings.Join(includes, ","), 
  "prints": strings.Join(prints, ","), 
 }

 file, err := os.Create("settings.txt")
 if err != nil {
  fmt.Println("Error creating file:", err)
  return
 }
 defer file.Close()
 for key, value := range settings {
  _, err := file.WriteString(fmt.Sprintf("%s=%s\n", key, value))
  if err != nil {
   fmt.Println("Error writing to file:", err)
  }

}

func loadSettings() {
 file, err := os.Open("settings.txt")
 if err != nil {
  fmt.Println("Error opening file:", err)
  return
 }
 defer file.Close()

 scanner := bufio.NewScanner(file)
 for scanner.Scan() {
  line := scanner.Text()
  parts := strings.SplitN(line, "=", 2)
  if len(parts) == 2 {
   key := strings.TrimSpace(parts[0])
   value := strings.TrimSpace(parts[1])

   switch key {
   case "private":
    privateOn = value == "true"
   case "timezone":
    if v, err := strconv.Atoi(value); err == nil {
     timezone = v
    }
   case "deleteOld":
    if v, err := strconv.Atoi(value); err == nil {
     deleteOld = v
    }
   case "timer":
    if v, err := strconv.Atoi(value); err == nil {
     timerSeconds = v
    }
   case "includes":
    includes = strings.Split(value, ",")
    for i := range includes {
     includes[i]= strings.TrimSpace(includes[i])
    }
   case "prints":
    prints = strings.Split(value, ",")
    for i := range prints {
     prints[i]= strings.TrimSpace(prints[i])
    }
   }
  }
 }
}

func saveLog(state int) string{
 content, err := os.ReadFile("log.txt")
 if err != nil {
  file, err:= os.Create("log.txt")
  if err != nil {
   fmt.Println("error creating log.txt")
   return ""
  }
  defer file.Close()
  _, err = file.WriteString(";recent\n")
  content = []byte(";recent\n")
 }
 lines := strings.Split(string(content), "\n")
 recentFound := false
 currentTime := time.Now().Unix()
 allFound:= false
 text:= windowName()
 if (state>0){
  if(state==2){
   allStrings:= []string{}
   for i:=0;i<len(lines);i++{
    if (lines[i]==";recent"){
     numbers := make([]int, len(allStrings))
     for j, str := range allStrings {
      parts := strings.Split(str, "=")
      if len(parts) == 2 {
       num, _ := strconv.Atoi(parts[1])
       numbers[j] = num
      }
     }
     index := make([]int, len(numbers))
     for k := range index {index[k] = k}
     sort.Slice(index, func(k, j int) bool {return numbers[index[k]] > numbers[index[j]]})
     sortedStrings := make([]string, len(allStrings))
     for k, idx := range index {sortedStrings[k] = allStrings[idx]}
     if len(sortedStrings) > 5 {sortedStrings = sortedStrings[:5]}
     newString:= strings.Join(sortedStrings, "\n")
     newString+= "\n"
     return newString
    }else{
     allStrings= append(allStrings, lines[i])
    }
   }
   return ""
  } //endState1
  logTime:= ""
  counter:=1
  customZone := time.FixedZone("CustomTZ", timezone*60*60)
  for i:=len(lines)-1;i>=0;i--{
   counter++
   if (lines[i]==";recent"){
    return logTime
   }
   times:= strings.Split(lines[i], "=")
   beginEnd:= strings.Split(times[1], "-")
   begin, _:= strconv.ParseInt(beginEnd[0], 10, 64)
   end, _:= strconv.ParseInt(beginEnd[1], 10, 64)
   beginEnd= []string{ time.Unix(begin,0).In(customZone).Format("15:04"), time.Unix(end,0).In(customZone).Format("15:04")}
   logTime+= fmt.Sprintf("%s= %s-%s\n", times[0], beginEnd[0], beginEnd[1])
   if(counter>logAmount){return logTime}
  }
 }
 for i:=0;i<len(lines);i++{
  if (!recentFound){
   if strings.Contains(lines[i], ";recent") {
    recentFound = true
   }else{
    if strings.Contains(lines[i], text){
     allFound= true
     parts := strings.Split(lines[i], "=")
     newTime, _ := strconv.Atoi(parts[1])
     newTime += timerSeconds
     lines[i] = parts[0] + "=" + strconv.Itoa(newTime)
    }
   }

  }else{
   rangeParts := strings.Split(lines[i], "-")
   endTime, err2 := strconv.ParseInt(rangeParts[len(rangeParts)-1], 10, 64)
   if err2 != nil {break}
   if currentTime- int64(deleteOld)> endTime {
    lines = append(lines[:i], lines[i+1:]...)
    i--
   }else{
    break
   }
  }
 }

 if(!allFound){
  lines = append([]string{text+ "="+ strconv.Itoa(timerSeconds)}, lines...)
 }
 if (len(lines)> 0){
  lastLine:= strings.Split(lines[len(lines)-1], "=")
  if (lastLine[0]== text&& len(lastLine)>1){
   times:= strings.Split(lastLine[1], "-")
   lines[len(lines)-1]= lastLine[0]+ "="+ times[0]+ "-"+ strconv.FormatInt(currentTime, 10)
  }else{
   newLine:= text + "=" + strconv.FormatInt(currentTime, 10) + "-" + strconv.FormatInt(currentTime, 10)
   lines = append(lines, newLine)
  }
 }
 file, err := os.Create("log.txt")
 defer file.Close()
 for i:=0;i<len(lines);i++ {
  if(i==len(lines)-1){
   file.WriteString(lines[i])
  }else{
   file.WriteString(lines[i] + "\n")
  }
 }
 return ""
}

  consoles:= strings.Split(console, ";")
  for i:= 0; i< len(commands); i++ {
   if strings.ToLower(consoles[0]) == commands[i] || aliases[i]== strings.ToLower(consoles[0]) {
    commandFunc(i, console)
    break
   }
   if i== len(commands)-1{
    fmt.Printf("no command: %s\n", console)
   }
  }
  console= ""
 }
}

func timerFunc() {
 ticker:= time.NewTicker(time.Duration(timerSeconds) * time.Second)
 defer ticker.Stop()
 for {
  select {
   case <-ticker.C:
    if debugOn {fmt.Println(windowName())}
    saveLog(0)
    previousTime = time.Now().Unix()
   case <-stopTicker:
    ticker.Stop()
    return
  }
 }
}

func runFile(nohup string){
 cmd := exec.Command("bash", "./run.sh", nohup)
 err := cmd.Start()
 if err != nil {
  fmt.Printf("run.sh is missing: %v\n", err)
 }else{
  fmt.Printf("ran run.sh (gnome default)\n")
 }
}

func trayBegin() {
 systray.SetIcon(iconData)
 systray.SetTitle("pieStatistic")
 systray.SetTooltip("pieStat")
 mAdd := systray.AddMenuItem("add", "add instance")
 systray.AddSeparator()
 mQuit := systray.AddMenuItem("quit", "remove instance")

 go func() {
  for {
   select {
   case <-mAdd.ClickedCh:
    runFile("false")
   case <-mQuit.ClickedCh:
    systray.Quit()
    return
   }
  }
 }()
}
